cmake_minimum_required(VERSION 3.21)
project(ruigo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Sql)
add_library(ruigo_core src/DictionaryRepository.cpp src/SearchHistory.cpp)
target_include_directories(ruigo_core PUBLIC src)
target_link_libraries(ruigo_core PUBLIC Qt6::Sql)
add_library(ruigo_ui src/MainWindow.cpp)
target_link_libraries(ruigo_ui PUBLIC ruigo_core Qt6::Widgets)
add_executable(ruigo WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(ruigo PRIVATE ruigo_ui)
set_target_properties(ruigo PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "jp.keishin.ruigo"
    MACOSX_BUNDLE_BUNDLE_NAME "Ruigo" MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(ruigo_core PRIVATE /utf-8)
    target_compile_options(ruigo_ui PRIVATE /utf-8)
    target_compile_options(ruigo PRIVATE /utf-8)
endif()
if(APPLE)
    set(RUIGO_DATA_DIR "$<TARGET_BUNDLE_DIR:ruigo>/Contents/Resources")
else()
    set(RUIGO_DATA_DIR "$<TARGET_FILE_DIR:ruigo>")
endif()
add_custom_target(ruigo_assets ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${RUIGO_DATA_DIR}/data"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROJECT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "${RUIGO_DATA_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/licenses" "${RUIGO_DATA_DIR}/licenses"
    DEPENDS ruigo)
if(EXISTS "${PROJECT_SOURCE_DIR}/data/dictionary.sqlite")
    add_custom_command(TARGET ruigo_assets POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${PROJECT_SOURCE_DIR}/data/dictionary.sqlite" "${RUIGO_DATA_DIR}/data/dictionary.sqlite")
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(ruigo_tests tests/Tests.cpp)
    target_link_libraries(ruigo_tests PRIVATE ruigo_ui Qt6::Test)
    target_compile_definitions(ruigo_tests PRIVATE RUIGO_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    if(WIN32 OR APPLE)
        add_test(NAME ruigo_tests COMMAND ruigo_tests -o test-results.txt,txt)
    else()
        add_test(NAME ruigo_tests COMMAND ruigo_tests -platform offscreen -o test-results.txt,txt)
    endif()
endif()
